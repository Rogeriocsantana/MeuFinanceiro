import {LoadingBar} from './Loading';

import {parseMoneyCents} from './formValues';
type Props={page:{key:string;name?:string};data:LocalBankData;nav:(key:string,name?:string)=>void;reload:()=>Promise<void>;month:number;picker:React.ReactNode;onMonth:(month:number)=>void};
export const livePages=['home','accounts','account','cards','transactions','category','analysis','income','assets','people','person','loans','loan','future'];

 function content(){if(['people','person','loans','loan','assets','future'].includes(page.key))return <LocalControlScreens page={page} data={data} nav={nav} picker={picker}>{page.key==='person'?list(all.filter(r=>data.annotations[r.id]?.personName===page.name||data.annotations[r.id]?.shares?.some(s=>s.person===page.name))):null}</LocalControlScreens>;switch(page.key){
  case 'home':return <><Card><Text style={u.h2}>Seu dinheiro</Text><Text style={u.small}>Bancos atualizados em {new Date(data.syncedAt).toLocaleString('pt-BR')}</Text>{picker}{metrics('Saldo em contas',accounts.reduce((n,a)=>n+a.amountCents,0))}<Text style={u.small}>Posição atual · independente do mês consultado</Text><View style={{flexDirection:'row',gap:16}}>{[['Receitas no mês',overview.income,C.green,'income'],['Despesas no mês',overview.expense,C.red,'analysis']].map(([label,value,color,key])=><TouchableOpacity key={String(key)} onPress={()=>nav(String(key))} style={{flex:1,gap:8}}><Text style={u.small}>{label}</Text><Text style={{fontSize:22,fontWeight:'600',color:String(color)}}>{money(Number(value))}</Text></TouchableOpacity>)}</View>{overview.unclassified>0&&<Text style={u.small}>Há {overview.unclassified} movimentos a classificar. Os totais serão ajustados conforme a classificação.</Text>}<Text style={u.small}>Cofrinhos são apresentados no detalhe da conta conforme o histórico disponível.</Text></Card><Card><Section title="Minhas contas" action="Ver contas" onPress={()=>nav('accounts')}/>{accounts.map(accountRow)}</Card><Card><Section title="Cartões"/>{cards.map(accountRow)}</Card><Card><Section title="Movimentos recentes" action="Ver todos" onPress={()=>nav('transactions')}/>{[...all].sort((a,b)=>b.date.localeCompare(a.date)).slice(0,5).map(row)}</Card></>;
  case 'accounts':return <><Text style={u.big}>Suas contas</Text><Card>{metrics('Saldo informado em contas',accounts.reduce((n,a)=>n+a.amountCents,0))}</Card><Card>{accounts.map(accountRow)}{!accounts.length&&<Text style={u.muted}>Nenhuma conta retornada.</Text>}</Card></>;
  case 'account':{const a=accounts.find(a=>a.accountId===page.name||a.name===page.name),reserves=a?accountReserves(data,a.accountId):[];return <><Text style={u.h2}>{a?accountTitle(a):'Conta'}</Text><Card>{a&&metrics('Disponível fora dos cofrinhos',a.amountCents)}<Text style={u.small}>Saldo informado pelo banco · {a?.date.split('-').reverse().join('/')}</Text></Card>{reserves.length>0&&<Card><Section title="Cofrinhos" action="Ver patrimônio" onPress={()=>nav('assets')}/>{reserves.map(g=><Row key={g.name} title={'Cofrinho '+g.name} subtitle="Ver movimentações reconhecidas" value={money(g.netCents)} icon="lock" onPress={()=>setReserve(reserve===g.name?'':g.name)}/>)}<Text style={u.small}>Líquido dos aportes e resgates no histórico importado. Pode diferir do saldo atual se faltarem movimentos ou rendimentos.</Text>{!!reserve&&reserves.find(g=>g.name===reserve)?.records.map(row)}</Card>}{picker}{list(periodRows.filter(r=>r.accountId===a?.accountId))}</>;}
  case 'cards':{const card=cards.find(c=>c.accountId===(selectedCard||page.name))||cards[0];if(!card)return <Card><Text style={u.muted}>Nenhum cartão retornado pela conexão.</Text></Card>;const bills=data.records.filter(r=>r.kind==='bill'&&r.accountId===card.accountId).sort((a,b)=>b.date.localeCompare(a.date)),bill=bills.find(b=>b.date.slice(0,7)===selectedMonth),rows=periodRows.filter(r=>r.accountId===card.accountId),forecast=rows.reduce((n,r)=>n-r.amountCents,0),total=bill?.amountCents??forecast,months=[Math.max(0,month-1),month,Math.min(11,month+1)],participations=new Map<string,number>();for(const tx of rows)for(const share of data.annotations[tx.id]?.shares||[])participations.set(share.person,(participations.get(share.person)||0)+parseMoneyCents(share.amount));return <><Text style={u.big}>Cartões</Text><View style={{flexDirection:'row',gap:10}}>{cards.map(c=><TouchableOpacity key={c.id} accessibilityRole="button" accessibilityState={{selected:c.id===card.id}} onPress={()=>setSelectedCard(c.accountId)} style={{flex:1,borderRadius:14,padding:14,borderWidth:1,borderColor:c.id===card.id?C.blue:C.line,backgroundColor:c.id===card.id?C.soft:'#fff',gap:8}}><InstitutionMark name={bankPresentation(c).identity.name}/><Text style={u.small}>{bankPresentation(c).title}</Text></TouchableOpacity>)}</View>{picker}<Card><Text style={u.h2}>{bankPresentation(card).title}</Text><BankInvoiceChart values={months.map(m=>bills.find(b=>b.date.slice(0,7)==='2026-'+String(m+1).padStart(2,'0'))?.amountCents??null)} labels={months.map(m=>new Date(2026,m,1).toLocaleString('pt-BR',{month:'short'}))} selected={1} onSelect={i=>onMonth(months[i]!)}/>{metrics(bill?'Fatura importada · '+selectedMonth:'Compras sincronizadas · '+selectedMonth,total)}<View style={{flexDirection:'row',gap:16}}><View style={{flex:1}}><Text style={u.small}>Fechamento</Text><Text style={u.rowValue}>{bill?.metadata.closingDate?.slice(0,10).split('-').reverse().join('/')||'Não informado'}</Text></View><View style={{flex:1}}><Text style={u.small}>Vencimento</Text><Text style={u.rowValue}>{bill?.date.split('-').reverse().join('/')||'Não informado'}</Text></View></View>{!bill&&<Text style={u.small}>Prévia dos movimentos disponíveis. Pode mudar até o fechamento.</Text>}<Text style={u.small}>Limite e outras faturas</Text>{card.metadata.creditLimit!==null&&<Text style={u.rowValue}>{money(Math.round(card.metadata.creditLimit*100))}</Text>}<Text style={u.small}>Saldo devedor total informado: {money(card.amountCents)}</Text></Card><Card><Text style={u.h2}>Quem paga esta fatura</Text>{metrics('Sua participação',Math.max(0,total-[...participations.values()].reduce((n,v)=>n+v,0)))}{[...participations].map(([name,value])=><Row key={name} title={name} subtitle="Participação classificada neste aparelho" value={money(value)} icon="users" onPress={()=>nav('person',name)}/>)}{!participations.size&&<Text style={u.small}>Classifique as compras para dividir entre pessoas. Vínculos do Apps Script ainda não foram migrados.</Text>}<Text style={u.small}>Compras parceladas ({rows.filter(r=>r.metadata.totalInstallments>1).length})</Text></Card>{list(rows)}</>;}
  case 'transactions':case 'category':return <><Text style={u.big}>Lançamentos</Text>{picker}{list(periodRows.filter(r=>page.key!=='category'||categoryName(data.annotations[r.id]?.categoryId)===page.name))}</>;
  case 'analysis':{const grouped=new Map<string,number>();for(const tx of periodRows.filter(r=>r.amountCents<0)){const annotation=data.annotations[tx.id]||{},shared=(annotation.shares||[]).reduce((n,s)=>n+(parseMoneyCents(s.amount)||0),0);const name=categoryName(annotation.categoryId);grouped.set(name,(grouped.get(name)||0)+Math.max(0,-tx.amountCents-shared));}const groups=[...grouped].filter(([,v])=>v>0),colors=['#205dff','#ed6862','#2faf8a','#edb83d','#a5a1df'];return <>{picker}<Card><Text style={u.h2}>Despesas por categoria</Text>{groups.length?<Donut values={groups.map(([,v])=>v)} colors={groups.map((_,i)=>colors[i%colors.length]!)} label="Sua participação" value={money(groups.reduce((n,[,v])=>n+v,0))}/>:<Text style={u.muted}>Nenhuma despesa neste mês.</Text>}{groups.map(([name,value])=><Row key={name} title={name} value={money(value)} icon="chart" onPress={()=>nav('category',name)}/>)}<Text style={u.small}>Classificações feitas no app. Categorias sugeridas pelo banco não substituem suas escolhas.</Text></Card></>;}
  case 'income':return <>{picker}<Card>{metrics('Entradas bancárias no período',sums(periodRows).income)}<Text style={u.small}>Inclui entradas ainda não classificadas. Transferências e devoluções precisam ser identificadas.</Text></Card>{list(periodRows.filter(r=>!r.metadata.card&&r.amountCents>0))}</>;
  case 'assets':return <><Text style={u.big}>Patrimônio</Text><Card>{metrics('Saldo em contas',accounts.reduce((n,a)=>n+a.amountCents,0))}</Card><Card><Text style={u.h2}>Cofrinhos reconhecidos</Text>{accounts.flatMap(a=>accountReserves(data,a.accountId).map(g=><Row key={a.id+g.name} title={g.name} subtitle={accountTitle(a)+' · líquido do histórico'} value={money(g.netCents)} icon="lock" onPress={()=>nav('account',a.accountId)}/>))}<Text style={u.small}>Aportes menos resgates do histórico disponível. Rendimentos e saldo inicial não informados não são presumidos.</Text></Card></>;
  case 'people':return <Card><Text style={u.h2}>Pessoas</Text>{personNames.map(name=><Row key={name} title={name} icon="users" onPress={()=>nav('person',name)}/>)}{!personNames.length&&<Text style={u.muted}>Associe uma pessoa ou participação a um lançamento.</Text>}</Card>;
  case 'person':return <><Text style={u.h2}>{page.name}</Text>{list(all.filter(r=>{const a=data.annotations[r.id];return a?.personName===page.name||a?.shares?.some(s=>s.person===page.name);}))}</>;
  default:return <Card><Text style={u.h2}>{page.key==='future'?'Futuro':'Empréstimos'}</Text><Text style={u.muted}>Nenhum registro local nesta área. A sincronização bancária não cria empréstimos ou vínculos automaticamente.</Text></Card>;
 }}

style={u.small}>{warning}</Text>)}{content()}<Modal visible={!!editing} transparent animationType="none" onRequestClose={()=>setEditing(null)}><SafeAreaView style={{flex:1,justifyContent:'flex-end',backgroundColor:'rgba(17,24,39,0.38)'}}><EnterMotion><View style={{position:'relative',maxHeight:740,backgroundColor:'#fff',padding:20,borderTopLeftRadius:22,borderTopRightRadius:22,gap:18}}><LoadingBar/><View style={{flexDirection:'row',alignItems:'center'}}><Text style={[u.h2,{flex:1}]}>Classificar lançamento</Text><IconButton icon="close" label="Fechar classificação" onPress={()=>setEditing(null)}/></View><ScrollView showsVerticalScrollIndicator={false} keyboardShouldPersistTaps="handled" contentContainerStyle={{gap:18}}><Text style={u.muted}>{editing?.name} · {money(editing?.amountCents||0)}</Text>{!!error&&<Text accessibilityRole="alert" style={{color:C.red}}>{error}</Text>}<ChoiceField label="Categoria" value={category||'Selecione'} people={[...new Set([...personNames,...(person.trim()?[person.trim()]:[])])]} value={shares} onChange={setShares} onCreatePerson={()=>setError('Informe o nome da nova pessoa no campo Pessoa para adicioná-la à lista.')}/>}<Text style={u.label}>Observações</Text><TextInput accessibilityLabel="Observações do lançamento" value={notes} onChangeText={setNotes} style={u.input}/></ScrollView><Button label="Salvar classificação" onPress={async()=>{if(!editing)return;const sum=shares.reduce((n,s)=>n+parseMoneyCents(s.amount),0);if(shares.some(s=>!s.person||s.person==='Selecione'||!(parseMoneyCents(s.amount)>0))||sum>Math.abs(editing.amountCents)){setError('Confira pessoas e valores das participações.');return;}const previous=data.annotations[editing.id]||{};const annotation:LocalAnnotation={...previous,categoryId:category==='Selecione'?'':categoryCatalog.find(c=>c.name===category)?.key||category,personId:person.trim()?previous.personName===person.trim()?previous.personId||person.trim():person.trim():undefined,personName:person.trim(),shares,notes,sourceAmountCentsAtEdit:editing.amountCents};await saveBankAnnotation(data.scope,editing.id,annotation);await reload();setEditing(null);}}/></View></EnterMotion></SafeAreaView></Modal></>;
